import { useState, useRef } from "react";
import { Nav, Form, Modal, Button } from "react-bootstrap";

import "../css/Nav.css";

function Login(props) {
	const [show, setShow] = useState(false);
	const handleClose = () => setShow(false);
	const handleShow = () => setShow(true);

	const idRef = useRef(null);
	const pwdRef = useRef(null);

	const [formData, setFormData] = useState({ id: "", pwd: "" });
	const [errors, setErrors] = useState({ id: "", pwd: "" });

	const handleFormChange = (e) => {
		const { name, value } = e.target;
		setFormData({ ...formData, [name]: value });
		setErrors({ id: "", pwd: "" }); // 입력 시 에러 초기화
	};

	// 로그인 제출 핸들러 (서버 없이  임시 버전)
	const handleLoginSubmit = async (e) => {
		e.preventDefault(); // 페이지 새로고침 방지

		if (!formData.id) {
			setErrors((prev) => ({ ...prev, id: "아이디를 입력해주세요" }));
			idRef.current.focus();
			return;
		}
		if (!formData.pwd) {
			setErrors((prev) => ({ ...prev, pwd: "비밀번호를 입력해주세요" }));
			pwdRef.current.focus();
			return;
		}

		// -------------------------------------------------------------
		// [임시 변경] 백엔드 서버(9000번 포트) 대용 가짜 로그인 로직
		// -------------------------------------------------------------
		if (formData.id === "test" && formData.pwd === "1234") {
			alert(`${formData.id}님, 로그인 되었습니다. (임시 모드)`);

			// 부모 상태 업데이트 함수 실행 (로그인 상태 반영)
			props.setIsLoggedIn(true);
			props.setUserId(formData.id);

			handleClose(); // 성공 시 모달창 닫기
		} else {
			alert("로그인에 실패하셨습니다.\n(테스트 아이디: test / 비밀번호: 1234)");
		}

		/* 
		try {
			// 백엔드 서버로 임시 로그인 요청
			const response = await fetch("http://localhost:9000/login", {
				method: "POST",
				headers: { "Content-Type": "application/json" },
				body: JSON.stringify(formData),
			});
			if (!response.ok) {
				console.error(`HTTP Error: ${response.status}`);
				if (response.status === 404) {
					alert(
						"서버가 localhost:9000 에서 실행 중인지 확인하세요.",
					);
				}
				return;
			}

			const result = await response.json();

			if (result.isLogin) {
				alert(`${formData.id}님, 로그인 되었습니다.`);

				props.setIsLoggedIn(true);
				props.setUserId(formData.id);

				handleClose(); 
			} else {
				alert("로그인에 실패하셨습니다.");
			}
		} catch (err) {
			console.error("서버 통신 에러:", err);
			alert("서버 연결에 실패했습니다.");
		}
		*/
	};

	return (
		<>
			<Nav.Link onClick={handleShow} href="#" className="loginLink">
				로그인
			</Nav.Link>
			<Modal show={show} onHide={handleClose} centered>
				<Modal.Header>
					<img src="/img/logo.svg" alt="logo" />
					<button onClick={handleClose} type="button">
						<i className="fa-solid fa-x"></i>
					</button>
				</Modal.Header>
				<Modal.Body>
					{/* 폼 전송 이벤트 연결 */}
					<Form onSubmit={handleLoginSubmit}>
						<Form.Group>
							<Form.Label>아이디</Form.Label>
							<Form.Control
								autoFocus
								name="id"
								value={formData.id}
								ref={idRef}
								onChange={handleFormChange}
							/>
							<span style={{ color: "red", fontSize: "0.8rem" }}>
								{errors.id}
							</span>
						</Form.Group>
						<Form.Group className="mt-3">
							<Form.Label>비밀번호</Form.Label>
							<Form.Control
								type="password"
								name="pwd"
								value={formData.pwd}
								ref={pwdRef}
								onChange={handleFormChange}
							/>
							<span style={{ color: "red", fontSize: "0.8rem" }}>
								{errors.pwd}
							</span>
						</Form.Group>

						<Button type="submit" className="loginBtn mt-4">
							로그인
						</Button>
					</Form>

					<div className="sign mt-3">
						<button type="button">회원가입</button>
						<button type="button">아이디·비밀번호 찾기</button>
					</div>
				</Modal.Body>
			</Modal>
		</>
	);
}

export default Login;
